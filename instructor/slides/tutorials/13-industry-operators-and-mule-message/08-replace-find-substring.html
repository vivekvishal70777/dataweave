Tiny example

%dw 2.0
import * from dw::core::Strings
output application/json
---
{
  redacted: replace(payload.notes, /INV-\d+/, "REDACTED"),
  ids: payload.notes find /INV-\d+/,
  family: substringBefore(payload.sku, "-"),
  trimmed: trim(payload.name)
}
