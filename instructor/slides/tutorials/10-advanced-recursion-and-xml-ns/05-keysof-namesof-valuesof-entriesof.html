Tiny example

%dw 2.0
output application/json
var obj = { a: 1, b: 2 }
---
{
  keys: keysOf(obj),       // ["a", "b"] as Keys
  names: namesOf(obj),     // ["a", "b"] as Strings
  values: valuesOf(obj),   // [1, 2]
  entries: entriesOf(obj)  // [{key: "a", value: 1}, ...]
}
